## Usage

```tsx
import {
  Form,
  Field,
  FieldLabel,
  FieldControl,
  FieldDescription,
  FieldValidity,
  FieldError,
  Fieldset,
  FieldsetLegend,
} from '@/registry/ui/form';

<Form>
  <Fieldset>
    <FieldsetLegend>Account</FieldsetLegend>
    <Field name="email">
      <FieldLabel>Email</FieldLabel>
      <FieldControl />
      <FieldDescription>We'll never share your email.</FieldDescription>
      <FieldError />
    </Field>
  </Fieldset>
</Form>;
```
