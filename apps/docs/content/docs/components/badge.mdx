---
title: Badge
description: A small status indicator for labelling and categorising content.
---

## Installation

```bash
npx shadcn@latest add "https://ora-ui.com/r/badge.json"
```

## Usage

```tsx
import { Badge } from '@/components/ui/badge';

<Badge>Badge</Badge>;
```

## Variants

The `variant` prop controls the visual style of the badge.

<ComponentPreview name="badge-variants">

```tsx
import { Badge } from '@/components/ui/badge';

export default function BadgeVariants() {
  return (
    <div className="flex flex-wrap gap-2">
      <Badge variant="solid">Solid</Badge>
      <Badge variant="outline">Outline</Badge>
      <Badge variant="surface">Surface</Badge>
      <Badge variant="soft">Soft</Badge>
    </div>
  );
}
```

</ComponentPreview>

## Theme

The `theme` prop applies a color scheme across all variants.

<ComponentPreview name="badge-theme">

```tsx
import { Badge } from '@/components/ui/badge';

export default function BadgeTheme() {
  return (
    <div className="flex flex-wrap gap-2">
      <Badge variant="solid" theme="gray">
        Gray
      </Badge>
      <Badge variant="solid" theme="accent">
        Accent
      </Badge>
      <Badge variant="solid" theme="destructive">
        Destructive
      </Badge>
    </div>
  );
}
```

</ComponentPreview>

## Examples

### With Icon

Icons can be placed inline alongside a label.

<ComponentPreview name="badge-with-icon">

```tsx
import { CheckCircleIcon, ClockIcon, EyeIcon } from '@heroicons/react/16/solid';
import { Badge } from '@/components/ui/badge';

export default function BadgeWithIcon() {
  return (
    <div className="flex flex-wrap gap-2">
      <Badge variant="soft" theme="warning">
        <ClockIcon />
        In Progress
      </Badge>
      <Badge variant="soft" theme="accent">
        <EyeIcon />
        In Review
      </Badge>
      <Badge variant="soft" theme="success">
        <CheckCircleIcon />
        Completed
      </Badge>
    </div>
  );
}
```

</ComponentPreview>

### Icon Only

Use `size="icon"` for compact status indicators where space is limited.

<ComponentPreview name="badge-icon-only">

```tsx
import { CheckCircleIcon, ClockIcon, EyeIcon } from '@heroicons/react/16/solid';
import { Badge } from '@/components/ui/badge';

export default function BadgeIconOnly() {
  return (
    <div className="flex flex-wrap gap-2">
      <Badge size="icon" variant="soft" theme="warning">
        <ClockIcon />
      </Badge>
      <Badge size="icon" variant="soft" theme="accent">
        <EyeIcon />
      </Badge>
      <Badge size="icon" variant="soft" theme="success">
        <CheckCircleIcon />
      </Badge>
    </div>
  );
}
```

</ComponentPreview>

### As a Link

Pass `render={<a />}` to render the badge as an anchor element.

<ComponentPreview name="badge-as-link">

```tsx
import { Badge } from '@/components/ui/badge';

export default function BadgeAsLink() {
  return <Badge render={<a href="/changelog" />}>New feature</Badge>;
}
```

</ComponentPreview>

## Props

### Badge

| Prop        | Type                                                            | Default     |
| ----------- | --------------------------------------------------------------- | ----------- |
| `variant`   | `"solid" \| "soft" \| "outline" \| "surface"`                   | `"soft"`    |
| `size`      | `"default" \| "icon"`                                           | `"default"` |
| `theme`     | `"gray" \| "accent" \| "destructive" \| "warning" \| "success"` | `"gray"`    |
| `render`    | `React.ReactElement`                                            | `<span>`    |
| `className` | `string`                                                        | —           |

All other props are forwarded to the rendered element.
