---
title: Toggle Group
description: A set of toggle buttons where one or more items can be pressed at a time.
---

## Installation

```bash
npx shadcn@latest add "https://ora-ui.com/r/toggle-group.json"
```

## Usage

```tsx
import { ToggleGroup, ToggleGroupItem } from '@/components/ui/toggle-group';

<ToggleGroup>
  <ToggleGroupItem value="a">A</ToggleGroupItem>
  <ToggleGroupItem value="b">B</ToggleGroupItem>
  <ToggleGroupItem value="c">C</ToggleGroupItem>
</ToggleGroup>;
```

## Examples

### Default

Items sit as separate toggles with a gap between them.

<ToggleGroup>
  <ToggleGroupItem value="edit" aria-label="Edit">
    <PencilSquareIcon />
  </ToggleGroupItem>
  <ToggleGroupItem value="preview" aria-label="Preview">
    <EyeIcon />
  </ToggleGroupItem>
  <ToggleGroupItem value="bookmark" aria-label="Bookmark">
    <BookmarkIcon />
  </ToggleGroupItem>
</ToggleGroup>

```tsx
import { BookmarkIcon, EyeIcon, PencilSquareIcon } from '@heroicons/react/16/solid';
import { ToggleGroup, ToggleGroupItem } from '@/components/ui/toggle-group';

export default function ToggleGroupDefault() {
  return (
    <ToggleGroup>
      <ToggleGroupItem value="edit" aria-label="Edit">
        <PencilSquareIcon />
      </ToggleGroupItem>
      <ToggleGroupItem value="preview" aria-label="Preview">
        <EyeIcon />
      </ToggleGroupItem>
      <ToggleGroupItem value="bookmark" aria-label="Bookmark">
        <BookmarkIcon />
      </ToggleGroupItem>
    </ToggleGroup>
  );
}
```

### Outline

Use `variant="outline"` with `spacing={0}` to render items as a connected pill with shared borders.

<ToggleGroup variant="outline" spacing={0}>
  <ToggleGroupItem value="edit" aria-label="Edit">
    <PencilSquareIcon />
  </ToggleGroupItem>
  <ToggleGroupItem value="preview" aria-label="Preview">
    <EyeIcon />
  </ToggleGroupItem>
  <ToggleGroupItem value="bookmark" aria-label="Bookmark">
    <BookmarkIcon />
  </ToggleGroupItem>
</ToggleGroup>

```tsx
import { BookmarkIcon, EyeIcon, PencilSquareIcon } from '@heroicons/react/16/solid';
import { ToggleGroup, ToggleGroupItem } from '@/components/ui/toggle-group';

export default function ToggleGroupOutline() {
  return (
    <ToggleGroup variant="outline" spacing={0}>
      <ToggleGroupItem value="edit" aria-label="Edit">
        <PencilSquareIcon />
      </ToggleGroupItem>
      <ToggleGroupItem value="preview" aria-label="Preview">
        <EyeIcon />
      </ToggleGroupItem>
      <ToggleGroupItem value="bookmark" aria-label="Bookmark">
        <BookmarkIcon />
      </ToggleGroupItem>
    </ToggleGroup>
  );
}
```

## Props

### ToggleGroup

| Prop          | Type                         | Default        |
| ------------- | ---------------------------- | -------------- |
| `variant`     | `"soft" \| "outline"`        | `"soft"`       |
| `size`        | `"sm" \| "md" \| "lg"`       | `"md"`         |
| `spacing`     | `number`                     | `0`            |
| `orientation` | `"horizontal" \| "vertical"` | `"horizontal"` |
| `className`   | `string`                     | —              |

### ToggleGroupItem

Inherits `variant` and `size` from the parent `ToggleGroup` context. Individual items can override either prop directly.

All other props are forwarded to the underlying [Base UI Toggle](https://base-ui.com/components/toggle) primitive.
